    protected override void ExecuteBehavior()
    {
        float elapsed = Time.time - timeOffset;
        float t = elapsed * frequency + phaseOffset;

        // Calculate new position based on pattern
        Vector3 newPosition = CalculatePosition(t);
        
        // Apply constant rotation if enabled
        if (constantRotationSpeed != 0f)
        {
            transform.Rotate(0f, 0f, constantRotationSpeed * Time.deltaTime);
        }
        
        // Rotate sprite to face movement direction
        if (rotateSprite && spriteRenderer != null)
        {
            Vector3 direction = newPosition - lastPosition;
            if (direction.sqrMagnitude > 0.0001f)
            {
                float angle = Mathf.Atan2(direction.y, direction.x) * Mathf.Rad2Deg;
                spriteRenderer.transform.rotation = Quaternion.Euler(0f, 0f, angle);
            }
        }

        lastPosition = transform.position;
        transform.position = newPosition;
    }

    private Vector3 CalculatePosition(float t)
    {
        float x = transform.position.x - moveSpeed * Time.deltaTime;
        float y = startY;

        switch (pattern)
        {
            case MovementPattern.Sine:
                y = startY + Mathf.Sin(t) * actualAmplitude;
                break;

            case MovementPattern.Cosine:
                y = startY + Mathf.Cos(t) * actualAmplitude;
                break;

            case MovementPattern.Circle:
                // Circular motion - both X and Y affected
                float circleX = Mathf.Cos(t) * actualAmplitude * 0.5f;
                x = transform.position.x - moveSpeed * Time.deltaTime + circleX * Time.deltaTime * frequency;
                y = startY + Mathf.Sin(t) * actualAmplitude;
                break;

            case MovementPattern.Figure8:
                // Figure-8 (Lissajous curve)
                y = startY + Mathf.Sin(t) * actualAmplitude;
                float figure8X = Mathf.Sin(t * 2f) * actualAmplitude * 0.3f;
                x = transform.position.x - moveSpeed * Time.deltaTime + figure8X * Time.deltaTime;
                break;

            case MovementPattern.Zigzag:
                // Sharp zigzag using triangle wave
                float triangleWave = Mathf.Abs((t % (Mathf.PI * 2f)) / Mathf.PI - 1f) * 2f - 1f;
                y = startY + triangleWave * actualAmplitude;
                break;
        }

        return new Vector3(x, y, transform.position.z);
    }

    }

        return pattern;
    }

        {
            float t = i * 0.2f + phaseOffset;
            float y = 0f;

            switch (pattern)
            {
                case MovementPattern.Sine:
                    y = Mathf.Sin(t) * previewAmplitude;
                    break;
                case MovementPattern.Cosine:
                    y = Mathf.Cos(t) * previewAmplitude;
                    break;
                case MovementPattern.Circle:
                    y = Mathf.Sin(t) * previewAmplitude;
                    xOffset = Mathf.Cos(t) * previewAmplitude * 0.3f;
                    break;
                case MovementPattern.Figure8:
                    y = Mathf.Sin(t) * previewAmplitude;
                    xOffset = Mathf.Sin(t * 2f) * previewAmplitude * 0.3f;
                    break;
                case MovementPattern.Zigzag:

                    break;
            }

            Vector3 pos = previewStart + new Vector3(-i * 0.3f + xOffset, y, 0f);

        }
