    public override void Kill()
    {
        if (!isActive) return;

        isActive = false;

        // Split when killed (if allowed)
        if (generation < maxGenerations)
        {
            Split();
        }

        OnEnemyKilled?.Invoke(this);
        DestroySelf();
    }

    private void Split()
    {
        // Calculate spawn positions
        float angleStep = 180f / (splitCount + 1);
        
        for (int i = 0; i < splitCount; i++)
        {
            // Calculate spread position
            float angle = angleStep * (i + 1);
            float yOffset = Mathf.Sin(angle * Mathf.Deg2Rad) * splitSpread;
            
            // Alternate above and below
            if (i % 2 == 0)
                yOffset = splitSpread * 0.5f * (i + 1) / splitCount;
            else
                yOffset = -splitSpread * 0.5f * (i + 1) / splitCount;

            Vector3 spawnPos = transform.position + new Vector3(splitHorizontalOffset, yOffset, 0f);

            // Spawn the split enemy
            GameObject splitEnemy = SpawnSplitEnemy(spawnPos);
            
            if (splitEnemy != null)
            {
                ConfigureSplitEnemy(splitEnemy);
            }
        }
    }

    private GameObject SpawnSplitEnemy(Vector3 position)
    {
        GameObject prefabToSpawn = splitPrefab != null ? splitPrefab : gameObject;
        
        // If using self as prefab, we need to instantiate before destroying
        if (splitPrefab == null)
        {
            // Create a copy with the same setup
            GameObject copy = Instantiate(gameObject, position, Quaternion.identity);
            return copy;
        }
        else
        {
            return Instantiate(splitPrefab, position, Quaternion.identity);
        }
    }

    private void ConfigureSplitEnemy(GameObject enemyObj)
    {
        // Scale down
        enemyObj.transform.localScale = transform.localScale * splitScaleMultiplier;

        // Configure the enemy component
        if (enemyObj.TryGetComponent<SplitEnemy>(out var splitEnemy))
        {
            splitEnemy.generation = this.generation + 1;
            splitEnemy.moveSpeed = this.moveSpeed * splitSpeedMultiplier;
            splitEnemy.birdTransform = this.birdTransform;
            splitEnemy.destroyX = this.destroyX;
            splitEnemy.maxGenerations = this.maxGenerations;
            splitEnemy.splitPrefab = this.splitPrefab;
            splitEnemy.splitCount = this.splitCount;
            splitEnemy.splitScaleMultiplier = this.splitScaleMultiplier;
            splitEnemy.splitSpeedMultiplier = this.splitSpeedMultiplier;
            splitEnemy.ApplyGenerationVisuals();
        }
        else if (enemyObj.TryGetComponent<BaseEnemy>(out var baseEnemy))
        {
            // If split prefab is a different enemy type
            baseEnemy.Initialize(moveSpeed * splitSpeedMultiplier, birdTransform, destroyX);
        }
    }

    private void ApplyGenerationVisuals()
    {
        if (spriteRenderer != null && generation < generationColors.Length)
        {
            spriteRenderer.color = generationColors[generation];
        }
    }

    /// <summary>
    /// Set the generation level
    /// </summary>
    public void SetGeneration(int gen)
    {
        generation = gen;
        ApplyGenerationVisuals();
    }

    /// <summary>
    /// Get current generation
    /// </summary>
    public int GetGeneration()
    {
        return generation;
    }

    /// <summary>
    /// Check if this enemy can still split
    /// </summary>
    public bool CanSplit()
    {
        return generation < maxGenerations;
    }
