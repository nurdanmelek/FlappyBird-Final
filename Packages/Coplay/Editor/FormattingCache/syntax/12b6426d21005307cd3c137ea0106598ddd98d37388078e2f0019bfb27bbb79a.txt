            float targetMultiplier = 1f + (maxScaleMultiplier - 1f) * distanceRatio;

            // Smoothly interpolate to target scale
            currentScaleMultiplier = Mathf.Lerp(currentScaleMultiplier, targetMultiplier, growSpeed * Time.deltaTime);

            // Check if fully grown
            isFullyGrown = currentScaleMultiplier >= maxScaleMultiplier * 0.95f;

            // Apply scale with optional pulse
            float finalMultiplier = currentScaleMultiplier;
            if (isFullyGrown && enablePulse)

                finalMultiplier += Mathf.Sin(Time.time * pulseSpeed) * pulseIntensity;
            }

            transform.localScale = initialScale * finalMultiplier;

            // Update color based on growth
            if (spriteRenderer != null)

                spriteRenderer.color = Color.Lerp(originalColor, growColor, distanceRatio);
            }
        }
        else if (isGrowing)
        {
            // Shrink back when bird moves away
            currentScaleMultiplier = Mathf.Lerp(currentScaleMultiplier, 1f, growSpeed * 0.5f * Time.deltaTime);
            transform.localScale = initialScale * currentScaleMultiplier;

            if (spriteRenderer != null)
            {
                spriteRenderer.color = Color.Lerp(spriteRenderer.color, originalColor, growSpeed * 0.5f * Time.deltaTime);
            }

            // Reset growing state when back to normal
            if (currentScaleMultiplier <= 1.05f)
            {
                isGrowing = false;
                isFullyGrown = false;
                currentScaleMultiplier = 1f;
                transform.localScale = initialScale;
                
                if (spriteRenderer != null)
                    spriteRenderer.color = originalColor;
            }
        }

    protected override void OnHitBird(Bird bird)
    {
        // Calculate total damage (base + bonus if grown)

        if (isGrowing && currentScaleMultiplier > 1.2f)
        {
            totalDamage += explosionBonusDamage;
        }

        // Deal damage
        for (int i = 0; i < totalDamage; i++)
        {
            bird.GetHit();
        }

        DestroySelf();
    }
