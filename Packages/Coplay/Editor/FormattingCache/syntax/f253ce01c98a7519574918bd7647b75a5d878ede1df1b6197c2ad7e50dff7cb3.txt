    /// <summary>
    /// Get enemy speed for a specific level
    /// </summary>
    public float GetEnemySpeed(int level)
    {
        float speed = baseSpeed + (level - 1) * speedIncreasePerLevel;
        return Mathf.Min(speed, maxSpeed);
    }

    /// <summary>
    /// Get a random enemy that can spawn at the given level
    /// </summary>
    public EnemySpawnEntry GetRandomEnemy(int level)
    {
        // Calculate total weight of available enemies
        float totalWeight = 0f;
        foreach (var entry in enemies)
        {
            if (entry.CanSpawnAtLevel(level) && entry.enemyPrefab != null)
            {
                totalWeight += entry.spawnWeight;
            }
        }

        if (totalWeight <= 0f) return null;

        // Random selection based on weight
        float random = Random.Range(0f, totalWeight);
        float currentWeight = 0f;

        foreach (var entry in enemies)
        {
            if (entry.CanSpawnAtLevel(level) && entry.enemyPrefab != null)
            {
                currentWeight += entry.spawnWeight;
                if (random <= currentWeight)
                {
                    return entry;
                }
            }
        }

        return null;
    }
