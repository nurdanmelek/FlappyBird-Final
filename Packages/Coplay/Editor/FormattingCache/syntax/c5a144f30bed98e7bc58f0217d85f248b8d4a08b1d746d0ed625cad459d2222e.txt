    /// <summary>
    /// Stop and clear all enemies
    /// </summary>
    public void ResetEnemySystem()
    {
        if (enemySpawner != null)
        {
            enemySpawner.StopAndClearAll();
        }

        ResetStatistics();
    }

    /// <summary>
    /// Pause all enemy movement
    /// </summary>
    public void PauseEnemies()
    {
        if (enemySpawner != null)
        {
            enemySpawner.PauseAllEnemies();
        }
    }

    /// <summary>
    /// Resume all enemy movement
    /// </summary>
    public void ResumeEnemies()
    {
        if (enemySpawner != null)
        {
            enemySpawner.ResumeAllEnemies();
        }
    }

    /// <summary>
    /// Set the current level for difficulty scaling
    /// </summary>
    public void SetLevel(int level)
    {
        if (enemySpawner != null)
        {
            enemySpawner.SetLevel(level);
        }
    }

    /// <summary>
    /// Get count of currently active enemies
    /// </summary>
    public int GetActiveEnemyCount()
    {
        return enemySpawner != null ? enemySpawner.GetActiveEnemyCount() : 0;
    }

    /// <summary>
    /// Get all active enemies
    /// </summary>
    public List<BaseEnemy> GetActiveEnemies()
    {
        return enemySpawner != null ? enemySpawner.GetActiveEnemies() : new List<BaseEnemy>();
    }

    /// <summary>
    /// Spawn a specific enemy at a position
    /// </summary>
    public BaseEnemy SpawnEnemy(GameObject prefab, Vector3 position)
    {
        return enemySpawner != null ? enemySpawner.SpawnEnemy(prefab, position) : null;
    }

    /// <summary>
    /// Kill all active enemies (for power-ups, etc.)
    /// </summary>
    public void KillAllEnemies()
    {
        if (enemySpawner == null) return;

        var enemies = enemySpawner.GetActiveEnemies();
        foreach (var enemy in enemies)
        {
            if (enemy != null)
            {
                enemy.Kill();
            }
        }
    }

    #endregion

    #region Statistics

    /// <summary>
    /// Get total enemies spawned this session
    /// </summary>
    public int GetTotalSpawned() => totalEnemiesSpawned;

    /// <summary>
    /// Get total enemies killed this session
    /// </summary>
    public int GetTotalKilled() => totalEnemiesKilled;

    /// <summary>
    /// Get total enemies that escaped (went off-screen)
    /// </summary>
    public int GetTotalEscaped() => totalEnemiesEscaped;

    /// <summary>
    /// Reset all statistics
    /// </summary>
    public void ResetStatistics()
    {
        totalEnemiesSpawned = 0;
        totalEnemiesKilled = 0;
        totalEnemiesEscaped = 0;
    }

    #endregion

    #region Event Handlers

    private void HandleEnemySpawned(BaseEnemy enemy)
    {
        totalEnemiesSpawned++;
        
        // Subscribe to kill event
        enemy.OnEnemyKilled += HandleEnemyKilled;
        
        OnEnemySpawned?.Invoke(enemy);
        OnEnemyCountChanged?.Invoke(GetActiveEnemyCount());
    }

    private void HandleEnemyDestroyed(BaseEnemy enemy)
    {
        // Check if it was killed or escaped
        // (If killed, OnEnemyKilled would have been called first)
        
        OnEnemyCountChanged?.Invoke(GetActiveEnemyCount());
    }

    private void HandleEnemyKilled(BaseEnemy enemy)
    {
        totalEnemiesKilled++;
        enemy.OnEnemyKilled -= HandleEnemyKilled;
        OnEnemyKilled?.Invoke(enemy);
    }

    #endregion
