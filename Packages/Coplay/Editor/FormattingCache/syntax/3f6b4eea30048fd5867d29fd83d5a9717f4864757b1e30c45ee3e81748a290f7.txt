    #region Virtual Methods

    /// <summary>
    /// Called when the enemy hits the bird
    /// </summary>
    protected virtual void OnHitBird(Bird bird)
    {
        // Default behavior: deal damage and destroy self
        for (int i = 0; i < damage; i++)
        {
            bird.GetHit();
        }
        
        DestroySelf();
    }

    /// <summary>
    /// Called when the enemy is killed (by player action, not bounds)
    /// </summary>
    public virtual void Kill()
    {
        if (!isActive) return;
        
        isActive = false;
        OnEnemyKilled?.Invoke(this);
        DestroySelf();
    }

    /// <summary>
    /// Check if enemy is out of bounds
    /// </summary>
    protected virtual void CheckBounds()
    {
        if (transform.position.x < destroyX)
        {
            DestroySelf();
        }
    }

    /// <summary>
    /// Destroy the enemy game object
    /// </summary>
    protected virtual void DestroySelf()
    {
        isActive = false;
        Destroy(gameObject);
    }

    #endregion

    #region Public Methods

    /// <summary>
    /// Initialize the enemy with runtime settings
    /// </summary>
    public virtual void Initialize(float speed, Transform bird, float destroyPosX)
    {
        moveSpeed = speed;
        birdTransform = bird;
        destroyX = destroyPosX;
        isActive = true;
    }

    /// <summary>
    /// Set the movement speed
    /// </summary>
    public void SetSpeed(float speed)
    {
        moveSpeed = speed;
    }

    /// <summary>
    /// Get the current movement speed
    /// </summary>
    public float GetSpeed()
    {
        return moveSpeed;
    }

    /// <summary>
    /// Set the damage value
    /// </summary>
    public void SetDamage(int dmg)
    {
        damage = dmg;
    }

    /// <summary>
    /// Pause/Resume enemy behavior
    /// </summary>
    public void SetActive(bool active)
    {
        isActive = active;
    }

    /// <summary>
    /// Check if enemy is currently active
    /// </summary>
    public bool IsActive()
    {
        return isActive;
    }

    #endregion

    #region Helper Methods

    /// <summary>
    /// Get distance to bird
    /// </summary>
    protected float GetDistanceToBird()
    {
        if (birdTransform == null) return float.MaxValue;
        return Vector2.Distance(transform.position, birdTransform.position);
    }

    /// <summary>
    /// Get direction to bird (normalized)
    /// </summary>
    protected Vector2 GetDirectionToBird()
    {
        if (birdTransform == null) return Vector2.left;
        return ((Vector2)(birdTransform.position - transform.position)).normalized;
    }

    #endregion
